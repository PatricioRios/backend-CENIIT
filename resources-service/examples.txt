# 7 Ejemplos de cURL para probar el endpoint de listar recursos (GET /v1/recursos)
#
# NOTA IMPORTANTE: Todos los endpoints requieren autenticación.
# Reemplaza <TU_TOKEN_JWT_AQUI> con un token válido de Keycloak.

# --------------------------------------------------------------------

# 1. Listado simple (sin filtros)
#
# Obtiene la primera página de recursos con el límite por defecto (ej: 10).
curl -X GET "http://localhost:8080/v1/recursos" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"

# --------------------------------------------------------------------

# 2. Filtrado por estado
#
# Obtiene todos los recursos que se encuentren en estado "MANTENIMIENTO".
curl -X GET "http://localhost:8080/v1/recursos?filter[estado][EQUALS]=MANTENIMIENTO" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"

# --------------------------------------------------------------------

# 3. Filtrado combinado con AND (por defecto)
#
# Obtiene los recursos cuyo nombre contiene la palabra "Laptop" Y cuyo estado es "ACTIVO".
curl -X GET "http://localhost:8080/v1/recursos?filter[nombre][CONTAINS]=Laptop&filter[estado][EQUALS]=ACTIVO" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"

# --------------------------------------------------------------------

# 4. Ordenamiento
#
# Obtiene todos los recursos ordenados por nombre de forma ascendente (A-Z).
curl -X GET "http://localhost:8080/v1/recursos?sort=nombre" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"

# --------------------------------------------------------------------

# 5. Paginación y ordenamiento combinado
#
# Obtiene la segunda página de resultados, con 20 por página, ordenados por fecha de actualización descendente.
curl -X GET "http://localhost:8080/v1/recursos?page=2&limit=20&sort=-updated_at" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"

# --------------------------------------------------------------------

# 6. Filtrado con AND (explícito)
#
# Es el comportamiento por defecto, pero se puede hacer explícito con el parámetro logical_op=AND.
# Busca recursos cuyo nombre NO contiene la palabra "Viejo" Y cuyo estado es "ACTIVO".
curl -X GET "http://localhost:8080/v1/recursos?filter[nombre][NOT_CONTAINS]=Viejo&filter[estado][EQUALS]=ACTIVO&logical_op=AND" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"

# --------------------------------------------------------------------

# 7. Filtrado con OR
#
# Busca recursos cuyo estado es "MANTENIMIENTO" O cuyo nombre contiene la palabra "Servidor".
# Se utiliza el parámetro ?logical_op=OR
curl -X GET "http://localhost:8080/v1/recursos?filter[estado][EQUALS]=MANTENIMIENTO&filter[nombre][CONTAINS]=Servidor&logical_op=OR" \
-H "Authorization: Bearer <TU_TOKEN_JWT_AQUI>"
